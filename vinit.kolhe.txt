let option =
document.createElement("option");

option.value = key;

option.textContent =
stages[crop][key];

stageSelect.appendChild(option);

}

}

// -------------------------
// CHECK SOIL
// -------------------------

function checkSoil(){


let farmer =
document.getElementById("farmerName").value.trim();

let village =
document.getElementById("village").value.trim();

let date =
document.getElementById("date").value;

let crop =
document.getElementById("crop").value;

let stage =
document.getElementById("stage").value;

let ph =
parseFloat(document.getElementById("ph").value);

let n =
parseFloat(document.getElementById("n").value);

let p =
parseFloat(document.getElementById("p").value);

let k =
parseFloat(document.getElementById("k").value);

let zn =
parseFloat(document.getElementById("zn").value);

let fe =
parseFloat(document.getElementById("fe").value);


// -------------------------
// VALIDATION
// -------------------------

if(
isNaN(ph) ||
isNaN(n) ||
isNaN(p) ||
isNaN(k) ||
isNaN(zn) ||
isNaN(fe)
){

if(language === "hi"){

alert("कृपया सभी soil test values भरें।");

}
else{

alert("Please enter all soil test values.");

}

return;

}


// -------------------------
// CROP NAME
// -------------------------

let cropNames = {

orange:"Orange",

wheat:"Wheat",

soybean:"Soybean",

chilli:"Chilli"

};


// -------------------------
// REPORT DETAILS
// -------------------------

document.getElementById("rFarmer").innerText =
farmer || "-";

document.getElementById("rVillage").innerText =
village || "-";

document.getElementById("rDate").innerText =
date || "-";

document.getElementById("rCrop").innerText =
cropNames[crop];

document.getElementById("rStage").innerText =
stages[crop][stage];


// -------------------------
// SOIL RESULT
// -------------------------

let result = "";

let lowN = n < 280;
let lowP = p < 10;
let lowK = k < 110;
let lowZn = zn < 0.6;
let lowFe = fe < 4.5;


// pH

if(ph < 6.5){

result +=
"🔴 Soil pH: Acidic<br>";

}
else if(ph <= 7.5){

result +=
"🟢 Soil pH: Normal<br>";

}
else{

result +=
"🟠 Soil pH: Alkaline<br>";

}


// N

if(lowN){

result +=
"🔴 Nitrogen (N): Low<br>";

}
else{

result +=
"🟢 Nitrogen (N): Normal<br>";

}


// P

if(lowP){

result +=
"🔴 Phosphorus (P): Low<br>";

}
else{

result +=
"🟢 Phosphorus (P): Normal<br>";

}


// K

if(lowK){

result +=
"🔴 Potassium (K): Low<br>";

}
else{

result +=
"🟢 Potassium (K): Normal<br>";

}


// Zn

if(lowZn){

result +=
"🔴 Zinc (Zn): Low<br>";

}
else{

result +=
"🟢 Zinc (Zn): Normal<br>";

}


// Fe

if(lowFe){

result +=
"🔴 Iron (Fe): Low<br>";

}
else{

result +=
"🟢 Iron (Fe): Normal<br>";

}


document.getElementById("result").innerHTML =
result;


// -------------------------
// SMART RECOMMENDATION
// -------------------------

let recommendation = "";


// HINDI

if(language === "hi"){

recommendation +=
"<b>फसल:</b> " +
cropNames[crop] +
"<br>";

recommendation +=
"<b>फसल की अवस्था:</b> " +
stages[crop][stage] +
"<br><br>";


recommendation +=
"<b>पोषक तत्वों पर ध्यान:</b><br><br>";


if(lowN){

recommendation +=
"🔴 Nitrogen कम है — नाइट्रोजन पोषण पर ध्यान दें।<br>";

}

if(lowP){

recommendation +=
"🔴 Phosphorus कम है — फॉस्फोरस पोषण पर ध्यान दें।<br>";

}

if(lowK){

recommendation +=
"🔴 Potassium कम है — पोटैशियम पोषण पर ध्यान दें।<br>";

}

if(lowZn){

recommendation +=
"🔴 Zinc कम है — जिंक की कमी को सुधारने पर ध्यान दें।<br>";

}

if(lowFe){

recommendation +=
"🔴 Iron कम है — आयरन की कमी को सुधारने पर ध्यान दें।<br>";

}


if(
!lowN &&
!lowP &&
!lowK &&
!lowZn &&
!lowFe
){

recommendation +=
"🟢 दिए गए मानों के अनुसार कोई प्रमुख कमी नहीं मिली।<br>";

}


recommendation +=
"<br><b>सामान्य सुझाव:</b><br>";


if(crop === "orange"){

recommendation +=
"🍊 Orange: इस अवस्था में balanced nutrition और उचित irrigation management पर ध्यान दें।<br>";

}

else if(crop === "wheat"){

recommendation +=
"🌾 Wheat: फसल की अवस्था के अनुसार balanced nutrition और irrigation management रखें।<br>";

}

else if(crop === "soybean"){

recommendation +=
"🌱 Soybean: balanced nutrition और उचित moisture management पर ध्यान दें।<br>";

}

else if(crop === "chilli"){

recommendation +=
"🌶️ Chilli: balanced NPK और micronutrient management पर ध्यान दें।<br>";

}


recommendation +=
"<br>⚠️ Exact fertilizer quantity soil-test report, soil type, crop stage और स्थानीय कृषि recommendation के आधार पर तय करें।";

}


// ENGLISH

else{

recommendation +=
"<b>Crop:</b> " +
cropNames[crop] +
"<br>";

recommendation +=
"<b>Crop Stage:</b> " +
stages[crop][stage] +
"<br><br>";


recommendation +=
"<b>Nutrients to focus on:</b><br><br>";


if(lowN){

recommendation +=
"🔴 Nitrogen is low — focus on nitrogen nutrition.<br>";

}

if(lowP){

recommendation +=
"🔴 Phosphorus is low — focus on phosphorus nutrition.<br>";

}

if(lowK){

recommendation +=
"🔴 Potassium is low — focus on potassium nutrition.<br>";

}

if(lowZn){

recommendation +=
"🔴 Zinc is low — focus on zinc deficiency management.<br>";

}

if(lowFe){

recommendation +=
"🔴 Iron is low — focus on iron deficiency management.<br>";

}


if(
!lowN &&
!lowP &&
!lowK &&
!lowZn &&
!lowFe
){

recommendation +=
"🟢 No major deficiency detected from the entered values.<br>";

}


recommendation +=
"<br><b>General Crop Suggestion:</b><br>";


if(crop === "orange"){

recommendation +=
"🍊 Orange: Focus on balanced nutrition and proper irrigation management at this stage.<br>";

}

else if(crop === "wheat"){

recommendation +=
"🌾 Wheat: Maintain balanced nutrition and proper irrigation according to crop stage.<br>";

}

else if(crop === "soybean"){

recommendation +=
"🌱 Soybean: Focus on balanced nutrition and proper moisture management.<br>";

}

else if(crop === "chilli"){

recommendation +=
"🌶️ Chilli: Focus on balanced NPK and micronutrient management.<br>";

}


recommendation +=
"<br>⚠️ Exact fertilizer quantity should be decided using the soil-test report, soil type, crop stage and local agricultural recommendations.";

}


document.getElementById("recommendation").innerHTML =
recommendation;


// -------------------------
// SHOW REPORT
// -------------------------

document.getElementById("report").style.display =
"block";


}
